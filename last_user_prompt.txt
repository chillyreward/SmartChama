'use client'
import { useState, useRef, 
  useEffect } from 'react'
import { responses } from 
  '@/data/chatbot-responses'

interface Message {
  id: number
  from: 'user' | 'bot'
  text: string
  time: string
}

function getResponse(input: string): 
  string {
  const lower = input.toLowerCase().trim()
  
  for (const r of responses) {
    if (r.triggers[0] === 'default') 
      continue
    if (r.triggers.some(t => 
      lower.includes(t)
    )) {
      return r.answer
    }
  }
  
  // Return default
  return responses.find(
    r => r.triggers[0] === 'default'
  )?.answer || 
  'I am not sure about that. ' +
  'Try asking about contributions, ' +
  'loans, or trust scores.'
}

export function ChatBot() {
  const [open, setOpen] = 
    useState(false)
  const [messages, setMessages] = 
    useState<Message[]>([
      {
        id: 0,
        from: 'bot',
        text: 'Hello! I am the SmartChama assistant. Ask me anything about contributions, loans, trust scores, or how the platform works.',
        time: new Date()
          .toLocaleTimeString(
            'en-KE', 
            { hour: '2-digit', 
              minute: '2-digit' }
          )
      }
    ])
  const [input, setInput] = 
    useState('')
  const [typing, setTyping] = 
    useState(false)
  const bottomRef = 
    useRef<HTMLDivElement>(null)
  const inputRef = 
    useRef<HTMLInputElement>(null)

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ 
      behavior: 'smooth' 
    })
  }, [messages])

  useEffect(() => {
    if (open) {
      setTimeout(() => {
        inputRef.current?.focus()
      }, 100)
    }
  }, [open])

  function handleSend() {
    const text = input.trim()
    if (!text) return

    const userMsg: Message = {
      id: Date.now(),
      from: 'user',
      text,
      time: new Date()
        .toLocaleTimeString(
          'en-KE',
          { hour: '2-digit', 
            minute: '2-digit' }
        )
    }

    setMessages(prev => [...prev, userMsg])
    setInput('')
    setTyping(true)

    // Simulate typing delay 
    // (300-800ms feels natural)
    const delay = 
      300 + Math.random() * 500

    setTimeout(() => {
      const answer = getResponse(text)
      const botMsg: Message = {
        id: Date.now() + 1,
        from: 'bot',
        text: answer,
        time: new Date()
          .toLocaleTimeString(
            'en-KE',
            { hour: '2-digit', 
              minute: '2-digit' }
          )
      }
      setMessages(prev => 
        [...prev, botMsg]
      )
      setTyping(false)
    }, delay)
  }

  return (

async function runFraudChecks(
  profileId: string,
  chamaId: string
) {
  const supabase = getSupabaseAdmin()
  
  // Check 1: Same phone in too many 
  // chamas (>5 is suspicious)
  const { data: memberships } = 
    await supabase
      .from('chama_memberships')
      .select('chama_id')
      .eq('profile_id', profileId)
  
  if ((memberships?.length || 0) > 5) {
    await flagFraud(
      profileId, null,
      'same_phone_multiple_chamas',
      `Member belongs to ${
        memberships?.length
      } groups which is unusually high`,
      'medium'
    )
  }
  
  // Check 2: Chama has fewer than 
  // 3 members after 30 days 
  // (possibly fake)
  const chamaCreated = await supabase
    .from('chamas_v2')
    .select('created_at')
    .eq('id', chamaId)
    .single()
  
  const daysSinceCreation = 
    (Date.now() - new Date(
      chamaCreated.data?.created_at
    ).getTime()) / (1000*60*60*24)
  
  const { count: memberCount } = 
    await supabase
      .from('chama_memberships')
      .select('*', { count: 'exact' })
      .eq('chama_id', chamaId)
      .eq('status', 'active')
  
  if (daysSinceCreation > 30 && 
      (memberCount || 0) < 3) {
    await flagFraud(
      null, chamaId,
      'fake_chama',
      'Group has fewer than 3 members ' +
      'after 30 days',
      'low'
    )
  }
  
  // Check 3: No real M-Pesa payments 
  // but high trust score attempt
  const { data: contribs } = 
    await supabase
      .from('contributions_v2')
      .select('mpesa_receipt, status')
      .eq('membership_id', profileId)
  
  const confirmedWithReceipt = 
    contribs?.filter(
      c => c.status === 'confirmed' && 
      c.mpesa_receipt
    ).length || 0
  
  // Trust score ONLY counts 
  // contributions with real 
  // M-Pesa receipts
  // This prevents fake self-recorded 
  // contributions from inflating score
  return confirmedWithReceipt
}

WITHDRAWAL CONSENT SYSTEM:
When admin tries to record a withdrawal:

1. Admin submits withdrawal request 
   with amount and reason
2. All active members receive 
   notification immediately
3. Members have 24 hours to vote 
   approve/reject
4. If majority approves (or no one 
   objects), withdrawal executes
5. If majority rejects, it is blocked
6. ALL withdrawal attempts 
   (approved or rejected) are 
   permanently logged

File: src/app/api/wallet/
  withdraw-request/route.ts

export async function POST(
  request: Request
) {
  const supabase = getSupabaseAdmin()
  const { chama_id, admin_id, 
    amount, reason } = 
    await request.json()
  
  // Get eligible voter count
  const { count } = await supabase
    .from('chama_memberships')
    .select('*', { count: 'exact' })
    .eq('chama_id', chama_id)
    .eq('status', 'active')
    .neq('profile_id', admin_id)
  
  // Create consent request
  const { data: consent } = 
    await supabase
      .from('withdrawal_consents')
      .insert({
        chama_id,
        requested_by: admin_id,
        amount,
        reason,
        total_eligible_voters: count,
        status: 'pending'
      })
      .select()
      .single()
  
  // Notify ALL members immediately
  const { data: members } = 
    await supabase
      .from('chama_memberships')
      .select('profile_id')
      .eq('chama_id', chama_id)
      .eq('status', 'active')
      .neq('profile_id', admin_id)
  
  for (const m of members || []) {
    await supabase
      .from('notifications')
      .insert({
        profile_id: m.profile_id,
        chama_id,
        type: 'withdrawal_request',
        title: 'Withdrawal Request',
        message: 
          `Your admin has requested ` +
          `to withdraw KSh ${
            amount.toLocaleString('en-KE')
          }. Reason: ${reason}. ` +
          `You have 24 hours to approve ` +
          `or reject this.`
      })
  }
  
  return Response.json({ 
    consent_id: consent.id 
  })
}
